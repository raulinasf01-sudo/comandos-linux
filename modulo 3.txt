#!/bin/bash

FECHA=$(date +"%d-%m-%Y-%H-%M")
ARCHIVO="backup-$FECHA.tar.gz"

tar -czvf $ARCHIVO /home/$USER

echo "Respaldo creado correctamente: $ARCHIVO"

chmod +x respaldo.sh
./respaldo.sh

ls

nano red.sh
#!/bin/bash

# Pedir nombre del archivo
echo "Ingrese el nombre que desea para el archivo:"
read NOMBRE

# Guardar resultado en el escritorio
ifconfig > /home/$USER/Escritorio/$NOMBRE.txt

echo "Archivo creado en el Escritorio con el nombre $NOMBRE.txt"

chmod +x red.sh

./red.sh
