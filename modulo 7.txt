# PRACTICA 1 – NFS (Servidor Linux + Cliente Linux)

# Instalar NFS en el Servidor
sudo dnf install nfs-utils -y    # RHEL/CentOS/Fedora
sudo systemctl enable --now nfs-server

# Crear carpeta OS3 con 100 archivos
sudo mkdir /srv/OS3
cd /srv/OS3
sudo touch Adrian{1..100}.txt

# Asignar permisos
sudo chown -R nobody:nobody /srv/OS3
sudo chmod -R 755 /srv/OS3

# Configurar exports
# Editar:
sudo nano /etc/exports
# Agregar:
/srv/OS3 *(rw,sync,no_root_squash)

# Aplicar:
sudo exportfs -ra

# Activar el firewall y agregar los puertos
sudo firewall-cmd --add-service=nfs --permanenet
sudo firewall-cmd --add-service=npc-bind --permanenet
sudo firewall-cmd --add-service=mountd --permanenet

# Cliente Linux: instalar NFS
sudo dnf install nfs-utils -y
sudo mkdir /mnt/nfs

# Descubrir export:
showmount -e <IP_del_servidor>

# Montar:
sudo mount <IP_del_servidor>:/srv/OS3 /mnt/nfs

# Ver archivos:
ls /mnt/nfs

# Montaje automático en /etc/fstab
# Editar:
sudo nano /etc/fstab
# Agregar al final:
<IP_del_servidor>:/srv/OS3   /mnt/nfs   nfs   defaults   0 0

# Probar sin reiniciar:
sudo mount -a

# Reiniciar y comprobar:
ls /mnt/nfs



PRACTICA 2 – SAMBA (Linux Server + Windows Cliente)

Instalar Samba
sudo dnf install samba samba-common samba-client -y
sudo systemctl enable --now smb nmb

Crear carpeta compartida
sudo mkdir /srv/samba
sudo chmod 777 /srv/samba

Crear archivos:
sudo touch Adrian{1..100}.txt

Crear usuario y grupo Samba
sudo useradd sambauser
sudo smbpasswd -a sambauser

Configurar /etc/samba/smb.conf
Editar:
sudo nano /etc/samba/smb.conf
Agregar al final:
[OS3_SHARE]
   path = /srv/samba
   writable = yes
   browsable = yes
   valid users = sambauser
   read only = no

Reiniciar Samba:
sudo systemctl restart smb

Activar los puertos
sudo firewall-cmd --add-port=137 --permanent
sudo firewall-cmd --add-port=139 --permanent
sudo firewall-cmd --add-port=445 --permanent
sudo firewall-cmd --reload

Desactivar la seguridad
sudo setenforce 0
